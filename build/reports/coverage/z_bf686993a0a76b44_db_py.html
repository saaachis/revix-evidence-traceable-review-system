1"""Engine and session management. 

2 

3One engine per process, created lazily so that importing the package does not 

4try to reach a database. Tests and the CLI both use session_scope; the API 

5uses get_session as a FastAPI dependency. 

6"""

8from __future__ import annotations 

9 

10from collections.abc import Iterator 

11from contextlib import contextmanager 

12from functools import lru_cache 

13 

14from sqlalchemy import Engine, create_engine 

15from sqlalchemy.orm import Session, sessionmaker 

16 

17from revix_core.settings import get_settings 

18 

19 

20@lru_cache 

21def get_engine() -> Engine: 

22 """One engine, with every wait bounded. 

23 

24 Every timeout here exists because its absence turns a fast, legible 

25 failure into a hang, and a hang is the worst failure mode a demo can have. 

26 /health is written to report an unreachable database as a 503, but without 

27 connect_timeout it never gets the chance: psycopg waits on the operating 

28 system's TCP timeout, the platform's health probe gives up first, and the 

29 instance is marked unhealthy with nothing in the log explaining why. 

30 """ 

31 settings = get_settings() 

32 return create_engine( 

33 settings.sync_database_url, 

34 echo=settings.db_echo, 

35 pool_size=settings.db_pool_size, 

36 # A little headroom over pool_size rather than a hard wall, so a burst 

37 # queues briefly instead of erroring. 

38 max_overflow=settings.db_pool_size, 

39 # Neon can be asleep, and a sleeping database answers the TCP 

40 # handshake without completing the connection. This bounds that. 

41 pool_pre_ping=True, 

42 pool_recycle=settings.db_pool_recycle_seconds, 

43 # How long a request waits for a free connection before giving up. 

44 # SQLAlchemy defaults to thirty seconds, which is far longer than any 

45 # reader will wait: under load we would rather refuse quickly than 

46 # accumulate a queue of browsers that have already left. 

47 pool_timeout=settings.db_pool_timeout_seconds, 

48 connect_args={"connect_timeout": settings.db_connect_timeout_seconds}, 

49 future=True, 

50 ) 

51

53@lru_cache 

54def get_sessionmaker() -> sessionmaker[Session]: 

55 return sessionmaker(bind=get_engine(), expire_on_commit=False, future=True)

58@contextmanager 

59def session_scope() -> Iterator[Session]: 

60 """Transactional scope. Commits on success, rolls back on any exception.""" 

61 session = get_sessionmaker()() 

62 try: 

63 yield session 

64 session.commit() 

65 except Exception: 

66 session.rollback() 

67 raise 

68 finally: 

69 session.close()

72def get_session() -> Iterator[Session]: 

73 """FastAPI dependency. Read-only by convention; the API never commits.""" 

74 session = get_sessionmaker()() 

75 try: 

76 yield session 

77 finally: 

78 session.close()
