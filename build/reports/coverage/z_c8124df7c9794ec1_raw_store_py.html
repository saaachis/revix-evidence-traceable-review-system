1"""Storing a fetched response, small enough that keeping it is affordable. 

2 

3The raw store exists so a parser can be improved and the evidence re-derived 

4without going back to the source. That is better engineering than re-fetching 

5and it is the politer thing to do to a site that let us read it once. 

6 

7What it is not is a reason to keep a third of a gigabyte of HTML. The bodies 

8were stored exactly as received, and 2,943 of them came to 410 MB, which is 

985% of a 512 MB database holding about 70 MB of actual product. The receipts 

10were six times the size of the thing they were receipts for, and the database 

11filled and stopped the pipeline for three nights. 

12 

13HTML and JSON are extremely compressible, which is the whole reason this file 

14is short. Roughly nine to one in practice on the pages we fetch, so the same 

15history costs tens of megabytes rather than hundreds and the promise the store 

16makes stays affordable. 

17""" 

18 

19from __future__ import annotations 

20 

21import gzip 

22 

23#: Written into RawPayload.content_encoding. Anything else, including null on 

24#: the rows stored before this existed, means the bytes are as received. 

25GZIP = "gzip" 

26 

27#: Level 6 is gzip's default and it is the right pick here. Level 9 spends 

28#: noticeably more CPU on every payload of every run to gain a couple of 

29#: percent on text that is already compressing ninefold, and this runs inside 

30#: a nightly with a timeout. 

31LEVEL = 6 

32 

33#: Below this, compression is not worth the header it adds. An empty body from 

34#: a skipped fetch would otherwise grow from nothing to twenty bytes. 

35MIN_BYTES = 512 

36 

37 

38def compress(body: bytes) -> tuple[bytes, str | None]: 

39 """Return the bytes to store and the encoding to record alongside them. 

40 

41 mtime=0 so the output depends only on the input. Without it gzip stamps 

42 the current time into the header, which would make two identical payloads 

43 compress to different bytes and quietly defeat anything that later 

44 compares stored rows. 

45 """ 

46 if len(body) < MIN_BYTES: 

47 return body, None 

48 packed = gzip.compress(body, compresslevel=LEVEL, mtime=0) 

49 # A body that grows under compression is already compressed, or is random. 

50 # Either way, store what we were given. 

51 if len(packed) >= len(body): 

52 return body, None 

53 return packed, GZIP 

54 

55 

56def decompress(body: bytes, content_encoding: str | None) -> bytes: 

57 """The inverse, tolerant of rows written before compression existed. 

58 

59 Those rows carry a null encoding and are returned untouched, which is what 

60 lets this be deployed without rewriting the history it inherits. 

61 """ 

62 if content_encoding == GZIP: 

63 return gzip.decompress(body) 

64 return body
