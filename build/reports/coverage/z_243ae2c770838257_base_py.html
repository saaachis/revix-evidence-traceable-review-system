1"""Declarative base and column conventions. 

2 

3Four schemas, split by lifecycle rather than by feature, per proposal 

4section 13: 

5 

6 raw immutable fetched payloads append only 

7 core canonical entities and evidence slowly changing 

8 analysis derived opinion and credibility recomputable 

9 serving materialised verdicts for the API fully derived 

10 

11The rule that makes this work: every table in analysis and serving must be 

12recomputable from raw and core alone. Nothing important is allowed to exist 

13only in a derived table. 

14""" 

15

24from sqlalchemy.orm import DeclarativeBase, Mapped, mapped_column

26SCHEMA_RAW = "raw" 

27SCHEMA_CORE = "core" 

28SCHEMA_ANALYSIS = "analysis" 

29SCHEMA_SERVING = "serving" 

30 

31ALL_SCHEMAS = (SCHEMA_RAW, SCHEMA_CORE, SCHEMA_ANALYSIS, SCHEMA_SERVING)

33# Predictable constraint names. Without this, Alembic autogenerate produces 

34# unnamed constraints that cannot be dropped in a downgrade. 

35NAMING_CONVENTION = { 

36 "ix": "ix_%(column_0_N_label)s", 

37 "uq": "uq_%(table_name)s_%(column_0_N_name)s", 

38 "ck": "ck_%(table_name)s_%(constraint_name)s", 

39 "fk": "fk_%(table_name)s_%(column_0_N_name)s", 

40 "pk": "pk_%(table_name)s", 

41}

43 

44class Base(DeclarativeBase): 

45 metadata = MetaData(naming_convention=NAMING_CONVENTION)

47 type_annotation_map = { 

48 dict[str, Any]: JSONB, 

49 list[str]: JSONB,

54def uuid_pk() -> Mapped[uuid.UUID]:
