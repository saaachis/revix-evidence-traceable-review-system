1"""Reading ownership signals out of free text. 

2 

3A review site hands you a "verified owner" flag and a distance field. Reddit 

4and YouTube hand you a paragraph. Section 12 of the proposal leans on 

5`ownership_duration_months` and `km_driven` heavily, and section 18.1 defines 

6the held-out gold set in terms of them, so on a platform that has no such 

7fields the only place they exist is in the sentence a person wrote. 

8 

9Deliberately conservative. A hint that is wrong is worse than a hint that is 

10missing, because a missing hint merely excludes a unit from the gold set 

11whereas a wrong one puts a two-week impression into the pool of long-term 

12ownership that the strategies are being scored against. 

13""" 

14 

15from __future__ import annotations

17import re

19# 45,000 km | 45000 kms | 45k kms | 1.2 lakh km | 90,000 kilometres 

20_KM = re.compile( 

21 r"(?<![\d.])(\d{1,3}(?:,\d{2,3})+|\d+(?:\.\d+)?)\s*(k|lakh|lac|l)?\s*" 

22 r"(?:km|kms|kilometer|kilometre|kilometers|kilometres)\b", 

23 re.IGNORECASE, 

24) 

25 

26# 2 years | 18 months | 2.5 yrs | two years 

27# 

28# Words as well as digits, because an owner review is prose. "Bought it two 

29# years ago" is one of the commonest ways these reviews state an ownership 

30# period, and reading only digits discarded every one of them silently. 

31_WORD_NUMBERS = { 

32 "one": 1, 

33 "two": 2, 

34 "three": 3, 

35 "four": 4, 

36 "five": 5, 

37 "six": 6, 

38 "seven": 7, 

39 "eight": 8, 

40 "nine": 9, 

41 "ten": 10, 

42 "eleven": 11, 

43 "twelve": 12, 

44} 

45_DURATION = re.compile( 

46 r"(?<![\d.])(\d+(?:\.\d+)?|" 

47 + "|".join(sorted(_WORD_NUMBERS, key=len, reverse=True)) 

48 + r")\s+(year|yr|yrs|years|month|months|mon|mos)\b", 

49 re.IGNORECASE, 

50) 

51 

52# Only count a duration that is actually about owning the thing. "waited 2 

53# months for delivery" and "18 months warranty" are not ownership periods. 

54_OWNERSHIP_CONTEXT = re.compile( 

55 r"\b(own|owned|owning|ownership|had|have had|driving|driven|riding|ridden|" 

56 r"using|used|bought|purchased|" 

57 r"with (?:the|my|this) (?:car|bike|scooter|vehicle))\b", 

58 re.IGNORECASE, 

59) 

60 

61_NOT_OWNERSHIP = re.compile( 

62 r"\b(warranty|waiting|waited|delivery|booked|booking|emi|loan|insurance|" 

63 r"service interval|free service)\b", 

64 re.IGNORECASE, 

65) 

66 

67# Beyond these a number is far likelier to be a price, a typo or a joke than a 

68# reading off an odometer. India's highest-mileage taxis do reach 500,000 km, 

69# but one of those in the pool would distort the gold set on its own. 

70_MAX_KM = 500_000 

71_MAX_MONTHS = 30 * 12

74def km_driven(text: str) -> int | None: 

75 """The largest distance the writer claims, or nothing. 

76 

77 Largest rather than first, because "bought it at 5,000 km, now at 60,000" 

78 is describing one odometer and the useful number is the second one. 

79 """ 

80 best: int | None = None 

81 for raw, unit in _KM.findall(text): 

82 value = float(raw.replace(",", "")) 

83 multiplier = {"k": 1_000, "lakh": 100_000, "lac": 100_000, "l": 100_000} 

84 if unit: 

85 value *= multiplier.get(unit.lower(), 1) 

86 if not (100 <= value <= _MAX_KM): 

87 continue 

88 km = round(value) 

89 if best is None or km > best: 89 ↛ 81line 89 didn't jump to line 81 because the condition on line 89 was always true

90 best = km 

91 return best

94def ownership_months(text: str) -> int | None: 

95 """How long they say they have had it, in months. 

96 

97 Requires an ownership word nearby and no contract word, so a warranty 

98 period or a waiting time is not mistaken for lived experience. 

99 """ 

100 best: int | None = None 

101 for match in _DURATION.finditer(text): 

102 window = text[max(0, match.start() - 60) : match.end() + 60] 

103 if _NOT_OWNERSHIP.search(window) or not _OWNERSHIP_CONTEXT.search(window): 

104 continue 

105 raw = match.group(1).lower() 

106 value = float(_WORD_NUMBERS[raw]) if raw in _WORD_NUMBERS else float(raw) 

107 unit = match.group(2).lower() 

108 months = round(value * 12) if unit.startswith(("year", "yr")) else round(value) 

109 if not (1 <= months <= _MAX_MONTHS): 109 ↛ 110line 109 didn't jump to line 110 because the condition on line 109 was never true

110 continue 

111 if best is None or months > best: 111 ↛ 101line 111 didn't jump to line 101 because the condition on line 111 was always true

112 best = months 

113 return best

116def looks_like_ownership_account(text: str) -> bool: 

117 """Whether the writer is speaking from ownership rather than opinion. 

118 

119 Not a verified-owner flag and must never be stored as one. It is a weak 

120 textual signal, used only to prefer first-hand accounts when a platform 

121 offers no verification at all. 

122 """ 

123 first_person = re.search(r"\b(i|my|we|our)\b", text, re.IGNORECASE) is not None 

124 return first_person and _OWNERSHIP_CONTEXT.search(text) is not None
