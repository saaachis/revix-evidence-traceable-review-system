84def redact_dsn(dsn: str) -> str: 

85 """Strip the password so a DSN is safe to print or paste into an issue.""" 

86 if "@" not in dsn or "//" not in dsn: 

87 return dsn 

88 scheme, _, rest = dsn.partition("//") 

89 creds, _, host = rest.rpartition("@") 

90 user = creds.split(":", 1)[0] if creds else "" 

91 return f"{scheme}//{user}:***@{host}" if user else f"{scheme}//{host}"

100@app.command("probe") 

101def probe( 

102 source: str = typer.Option(..., "--source", "-s", help="Connector source_key."), 

103 manufacturer: str = typer.Option("Hyundai", "--manufacturer"), 

104 model: str = typer.Option("Creta", "--model"), 

105 variant: str = typer.Option("SX (O) Turbo DCT", "--variant"), 

106 vehicle_class: str = typer.Option("car", "--class", help="car or two_wheeler."), 

107) -> None: 

108 """Does this source still work? Answers without touching the database. 

109 

110 Runs one connector's discover, fetch and parse for a single vehicle and 

111 reports what came back. Nothing is written anywhere. 

112 

113 Two jobs. Checking a credential without granting anyone database access, 

114 and finding out whether a site has changed its markup before a nightly run 

115 quietly returns nothing at four in the morning. 

116 """ 

117 try: 

118 connector = registry.get(source) 

119 except KeyError as exc: 

120 typer.secho(str(exc), fg="red") 

121 raise typer.Exit(1) from exc 

122 

123 seed = CatalogSeed( 

124 variant_id="probe", 

125 manufacturer=manufacturer, 

126 model=model, 

127 variant_name=variant, 

128 vehicle_class=vehicle_class, 

129 ) 

130 typer.secho(f"{source}: {manufacturer} {model} ({vehicle_class})", bold=True) 

131 

132 start = time.monotonic() 

133 try: 

134 refs = list(connector.discover(seed)) 

135 except Exception as exc: 

136 typer.secho(f" discover failed: {type(exc).__name__}: {exc}", fg="red") 

137 raise typer.Exit(1) from exc 

138 

139 typer.echo(f" discovered {len(refs):>6,}") 

140 if not refs: 

141 typer.secho( 

142 " nothing to fetch. Either this source skips this vehicle class, " 

143 "or its search returned no results.", 

144 fg="yellow", 

145 ) 

146 raise typer.Exit(1) 

147 

148 drafts = [] 

149 failures = 0 

150 for ref in refs: 

151 try: 

152 drafts.extend(connector.parse(connector.fetch(ref))) 

153 except Exception as exc: 

154 failures += 1 

155 typer.secho(f" {ref.url}: {type(exc).__name__}: {exc}", fg="yellow") 

156 

157 typer.echo(f" fetched {len(refs) - failures:>6,}") 

158 typer.echo(f" parsed into units {len(drafts):>6,}") 

159 typer.echo( 

160 f" with a rating {sum(1 for d in drafts if d.rating_raw is not None):>6,}" 

161 ) 

162 typer.echo(f" with a date {sum(1 for d in drafts if d.published_at):>6,}") 

163 typer.echo(f" naming a variant {sum(1 for d in drafts if d.variant_hint):>6,}") 

164 typer.echo( 

165 f" stating ownership {sum(1 for d in drafts if d.ownership_duration_months):>6,}" 

166 ) 

167 typer.echo(f" took {time.monotonic() - start:>6.1f}s") 

168 

169 if drafts: 

170 sample = drafts[0] 

171 typer.secho("\n a sample unit", fg="cyan") 

172 typer.echo(f" listing {sample.listing_title}") 

173 typer.echo(f" rating {sample.rating_raw} / {sample.rating_scale_max}") 

174 typer.echo(f" published {sample.published_at}") 

175 typer.echo(f" text {sample.text[:120].replace(chr(10), ' ')}...") 

176 

177 # Reaching a source and getting nothing usable out of it is a failure, and 

178 # a scheduled check that exits 0 on that would be worse than no check. 

179 if not drafts: 

180 typer.secho("\n fetched, but parsed nothing. The markup may have changed.", fg="red") 

181 raise typer.Exit(1)

184@catalogue_app.command("discover") 

185def catalogue_discover( 

186 candidates: str = typer.Option("data/seed/candidates.json", "--candidates"), 

187 catalogue: str = typer.Option("data/seed/catalogue.json", "--catalogue"), 

188 dry_run: bool = typer.Option(False, "--dry-run", help="Report, write nothing."), 

189) -> None: 

190 """Read variant specs from the sources and add them to the catalogue. 

191 

192 Run by hand, never in the nightly. A catalogue that changes under a 

193 running pipeline is a catalogue nobody can reason about. 

194 

195 Specs come from the same sites the reviews do, under the same permission: 

196 CarDekho publishes a schema.org ProductGroup per model and a Car object 

197 per variant carrying fuel, gearbox, displacement and the ARAI figure. 

198 Typing those by hand would scale badly and, worse, would mean a spec 

199 somebody half-remembered. 

200 """ 

201 from revix_pipeline.catalogue_discovery import Candidate, discover 

202 from revix_pipeline.connectors.politeness import PoliteClient 

203 

204 src = json.loads(pathlib.Path(candidates).read_text(encoding="utf-8")) 

205 book = json.loads(pathlib.Path(catalogue).read_text(encoding="utf-8")) 

206 known_models = {m["slug"] for m in book["models"]} 

207 known_makes = {m["slug"] for m in book["manufacturers"]} 

208 # A model refers to its manufacturer by slug, not by display name. Writing 

209 # the name here produced a catalogue that looked right and blew up in the 

210 # seeder with KeyError: 'Maruti Suzuki'. 

211 slug_for_make = {m["name"]: m["slug"] for m in book["manufacturers"]} 

212 slug_for_make.update({m["name"]: m["slug"] for m in src.get("manufacturers", [])}) 

213 

214 client = PoliteClient("catalogue", rate_limit_rpm=10, respect_robots=True) 

215 added_models: list[dict[str, Any]] = [] 

216 empty: list[str] = [] 

217 start = time.monotonic() 

218 

219 for row in src["candidates"]: 

220 if row["slug"] in known_models: 

221 continue 

222 candidate = Candidate(**{k: v for k, v in row.items() if k != "notes"}) 

223 variants = discover(candidate, client) 

224 label = f"{candidate.manufacturer} {candidate.name}" 

225 if not variants: 

226 # Recorded rather than skipped silently. A model that yields 

227 # nothing is a slug that has moved, and finding that out three 

228 # weeks later from an empty page is the expensive way. 

229 empty.append(label) 

230 typer.secho(f" {label:34} nothing found", fg="yellow") 

231 continue 

232 typer.secho(f" {label:34} {len(variants)} variants", fg="green") 

233 make_slug = slug_for_make.get(candidate.manufacturer) 

234 if make_slug is None: 

235 typer.secho( 

236 f" {label:34} unknown manufacturer, add it to the candidates file", 

237 fg="red", 

238 ) 

239 empty.append(label) 

240 continue 

241 added_models.append( 

242 { 

243 "manufacturer": make_slug, 

244 "slug": candidate.slug, 

245 "name": candidate.name, 

246 "vehicle_class": candidate.vehicle_class, 

247 "body_style": candidate.body_style, 

248 "segment": candidate.segment, 

249 "launch_year": candidate.launch_year, 

250 "variants": [v.as_dict() for v in variants], 

251 } 

252 ) 

253 

254 new_makes = [m for m in src.get("manufacturers", []) if m["slug"] not in known_makes] 

255 total_variants = sum(len(m["variants"]) for m in added_models) 

256 

257 typer.secho( 

258 f"\n{len(added_models)} models, {total_variants} variants, " 

259 f"{len(new_makes)} new manufacturers, in {time.monotonic() - start:.0f}s", 

260 bold=True, 

261 ) 

262 if empty: 

263 typer.secho(f" {len(empty)} yielded nothing: {', '.join(empty)}", fg="yellow") 

264 

265 if dry_run: 

266 typer.echo(" dry run, nothing written") 

267 return 

268 if not added_models: 

269 typer.secho(" nothing new to add", fg="yellow") 

270 return 

271 

272 book["manufacturers"].extend(new_makes) 

273 book["models"].extend(added_models) 

274 pathlib.Path(catalogue).write_text( 

275 json.dumps(book, indent=2, ensure_ascii=False) + "\n", encoding="utf-8" 

276 ) 

277 grand = sum(len(m.get("variants", [])) for m in book["models"]) 

278 typer.secho( 

279 f" {catalogue} now holds {len(book['models'])} models and {grand} variants", 

280 fg="green", 

281 bold=True, 

282 ) 

283 typer.secho(" now run: uv run revix catalogue seed", fg="cyan")

345def model_train( 

346 out: str = typer.Option("data/models/aspect_classifier.joblib", "--out"), 

347 gold_path: str = typer.Option("data/gold/aspects.jsonl", "--gold"), 

348) -> None: 

349 """Train the aspect classifier on weak labels from the lexicon.

368 path = classifier.save(pathlib.Path(out)) 

369 typer.secho( 

370 f"trained in {time.monotonic() - start:.1f}s, written to {path}", fg="green", bold=True

374@model_app.command("evaluate") 

375def model_evaluate( 

376 gold_path: str = typer.Option("data/gold/aspects.jsonl", "--gold"), 

377 model_path: str = typer.Option("data/models/aspect_classifier.joblib", "--model"), 

378 record: bool = typer.Option(False, "--record", help="Write the result to eval_run."), 

379) -> None: 

380 """Score the lexicon and the classifier on the same hand-labelled set. 

381 

382 Reports macro F1 as the headline. Micro is dominated by the common topics, 

383 so a system that handles looks and mileage well and the service centre 

384 badly scores respectably on it; macro treats every topic equally and is 

385 the number that notices. 

386 """ 

387 from revix_pipeline.ml.aspect_model import AspectClassifier, evaluate_against_gold 

388 from revix_pipeline.ml.gold import load_gold 

389 

390 gold = load_gold(pathlib.Path(gold_path)) 

391 labelled = [item for item in gold if item.is_labelled] 

392 if not labelled: 

393 typer.secho( 

394 f"nothing labelled in {gold_path}. A classifier with no gold set is a " 

395 "lexicon with extra steps, so this refuses to print a number. " 

396 " Start here: docs/labelling-the-gold-set.md", 

397 fg="red", 

398 ) 

399 raise typer.Exit(1) 

400 

401 classifier = AspectClassifier.load(pathlib.Path(model_path)) 

402 results = evaluate_against_gold(gold, classifier) 

403 

404 typer.secho(f"measured on {len(labelled):,} hand-labelled sentences", bold=True) 

405 typer.echo(f" {'system':14} {'macro F1':>9} {'micro F1':>9}") 

406 for result in results: 

407 typer.echo(f" {result.name:14} {result.macro_f1:>9.3f} {result.micro_f1:>9.3f}") 

408 

409 if len(results) == 2: 

410 delta = results[1].macro_f1 - results[0].macro_f1 

411 colour = "green" if delta > 0 else "yellow" 

412 verdict = "better than" if delta > 0 else "no better than" 

413 typer.secho( 

414 f"\n the classifier is {verdict} the lexicon by {delta:+.3f} macro F1", fg=colour 

415 ) 

416 

417 typer.secho("\n per topic, best system", fg="cyan") 

418 best = results[-1] 

419 for aspect, scores in sorted(best.per_aspect.items(), key=lambda kv: -kv[1]["f1"]): 

420 typer.echo( 

421 f" {aspect:22} F1 {scores['f1']:.3f} " 

422 f"P {scores['precision']:.3f} R {scores['recall']:.3f} n={int(scores['support'])}" 

423 ) 

424 

425 if record: 

426 with session_scope() as session: 

427 for result in results: 

428 session.add( 

429 EvalRun( 

430 component="aspect_extraction", 

431 system=result.name, 

432 git_sha=_git_sha(), 

433 n_items=result.n_items, 

434 primary_metric="macro_f1", 

435 primary_value=round(result.macro_f1, 4), 

436 detail=result.as_dict(), 

437 ) 

438 ) 

439 typer.secho(f"\n {len(results)} result(s) recorded to eval_run", fg="green")

459 """Confirm the database is reachable and the extensions are installed.""" 

460 settings = get_settings() 

461 with session_scope() as session:

463 extensions = sorted(

512@db_app.command("purge") 

513def db_purge( 

514 source: list[str] = typer.Option( 

515 ..., "--source", "-s", help="Source key to remove. Repeat for several." 

516 ), 

517 yes: bool = typer.Option(False, "--yes", help="Skip the confirmation prompt."), 

518) -> None: 

519 """Remove every trace of a source, and the verdicts built on it. 

520 

521 Written for one specific job: replacing generated development evidence 

522 with real reviews, without hand-editing a production database. 

523 

524 Deleting the source row cascades to its raw payloads, ingest runs, 

525 listings and evidence units, and from those to aspect opinions and claim 

526 citations. Verdicts go too. They are fully derived, recomputing them costs 

527 one `revix enrich fuse`, and leaving them would publish scores built on 

528 evidence that no longer exists, with citations pointing at deleted rows. 

529 

530 The catalogue and the reference data are untouched. 

531 """ 

532 with session_scope() as session: 

533 rows = session.execute( 

534 select(EvidenceSource.id, EvidenceSource.source_key, EvidenceSource.display_name).where( 

535 EvidenceSource.source_key.in_(source) 

536 ) 

537 ).all() 

538 found = {r.source_key for r in rows} 

539 missing = sorted(set(source) - found) 

540 if missing: 

541 typer.secho(f"unknown source(s): {', '.join(missing)}", fg="red") 

542 raise typer.Exit(1) 

543 

544 units = session.scalar( 

545 select(func.count()) 

546 .select_from(EvidenceUnit) 

547 .where(EvidenceUnit.source_id.in_([r.id for r in rows])) 

548 ) 

549 verdicts = session.scalar(select(func.count()).select_from(Verdict)) 

550 

551 typer.secho("This will permanently delete:", fg="yellow", bold=True) 

552 for row in rows: 

553 typer.echo(f" {row.source_key:20} {row.display_name}") 

554 typer.echo(f" {units:,} evidence units, and every opinion and citation from them") 

555 typer.echo(f" {verdicts:,} verdicts, which must be recomputed afterwards") 

556 

557 if not yes and not typer.confirm("\nProceed?"): 

558 typer.echo("nothing was deleted.") 

559 raise typer.Exit(1) 

560 

561 session.execute(delete(Verdict))

654@db_app.command("prune-raw") 

655def db_prune_raw( 

656 keep_days: int = typer.Option( 

657 None, "--keep-days", help="Override settings.raw_retention_days." 

658 ), 

659 max_mb: int = typer.Option(None, "--max-mb", help="Override settings.raw_max_megabytes."), 

660 batch_size: int = typer.Option( 

661 50, 

662 "--batch-size", 

663 help="Payloads per transaction. Smaller survives a tighter squeeze.", 

664 ), 

665 dry_run: bool = typer.Option(False, "--dry-run", help="Say what would go, delete nothing."), 

666) -> None: 

667 """Hold the raw store inside a byte budget, reclaiming space as it goes. 

668 

669 Two rules decide what goes: a payload is surplus if it is older than the 

670 retention window, or if keeping it would push the store past its size 

671 budget, oldest first. The age rule alone was the first attempt and it 

672 freed nothing, because every payload was inside the window. An age window 

673 bounds how old a store gets; it does not bound how big it gets, and size 

674 was what broke. 

675 

676 It deletes in batches, and that is not a tuning knob, it is the only way 

677 this works at all. Deleting the surplus in one statement failed with the 

678 very error it was meant to fix: a full database has no room to record a 

679 large deletion. Two things make it expensive. Dead rows are written, not 

680 just marked absent, and evidence_unit.raw_payload_id is ON DELETE SET 

681 NULL, so every payload removed also rewrites the evidence rows pointing at 

682 it. Fifty at a time, vacuuming after each, spends the small amount of 

683 headroom that exists to create more. 

684 

685 Only raw.raw_payload is touched. Every evidence unit stays exactly where 

686 it is and keeps its text: it loses a pointer to a receipt, nothing else. 

687 

688 One subtlety worth knowing when reading the numbers. The budget is 

689 measured with length(body), which is the logical size, while Postgres 

690 stores these compressed out of line, so the figures here run larger than 

691 the disk they occupy. Sizing against the logical number is the safe 

692 direction to be wrong in. 

693 """ 

694 settings = get_settings() 

695 days = keep_days if keep_days is not None else settings.raw_retention_days 

696 budget_mb = max_mb if max_mb is not None else settings.raw_max_megabytes 

697 cutoff = datetime.now(UTC) - timedelta(days=days) 

698 budget = budget_mb * 1024 * 1024 

699 

700 # A running total over the rows newest-first: everything past the budget is 

701 # surplus. Resolved once, into a list, because deleting changes the running 

702 # totals and re-running this between batches would keep moving the target. 

703 doomed_sql = text( 

704 """ 

705 select id, length(body) as bytes from ( 

706 select id, body, fetched_at, 

707 sum(length(body)) over (order by fetched_at desc, id) as running 

708 from raw.raw_payload 

709 ) ranked 

710 where ranked.fetched_at < :cutoff or ranked.running > :budget 

711 order by ranked.fetched_at 

712 """ 

713 ) 

714 

715 with session_scope() as session: 

716 total, total_bytes = session.execute( 

717 text("select count(*), coalesce(sum(length(body)), 0) from raw.raw_payload") 

718 ).one() 

719 doomed = session.execute(doomed_sql, {"cutoff": cutoff, "budget": budget}).all() 

720 

721 freeing = sum(int(row[1]) for row in doomed) 

722 typer.echo(f" held now {total:>8,} ({_human(int(total_bytes))} logical)") 

723 typer.echo(f" budget {budget_mb:>8,} MB, keeping {days} days") 

724 typer.echo(f" to delete {len(doomed):>8,} ({_human(freeing)})") 

725 typer.echo(f" to keep {total - len(doomed):>8,} ({_human(int(total_bytes) - freeing)})") 

726 

727 if dry_run: 

728 typer.secho("dry run, nothing deleted", fg="yellow") 

729 return 

730 if not doomed: 

731 typer.secho("already inside the budget", fg="green") 

732 return 

733 

734 ids = [row[0] for row in doomed] 

735 engine = get_engine() 

736 done = 0 

737 for start in range(0, len(ids), batch_size): 

738 chunk = ids[start : start + batch_size] 

739 with session_scope() as session: 

740 session.execute( 

741 text("delete from raw.raw_payload where id = any(:ids)"), {"ids": chunk} 

742 ) 

743 # After each batch, not at the end. Deleting marks rows dead; it does 

744 # not hand the pages back, and the next batch needs the room this one 

745 # released. VACUUM cannot run inside a transaction, hence autocommit. 

746 # evidence_unit is vacuumed too because the SET NULL rewrote rows there. 

747 with engine.connect().execution_options(isolation_level="AUTOCOMMIT") as conn: 

748 conn.execute(text("vacuum raw.raw_payload")) 

749 conn.execute(text("vacuum core.evidence_unit")) 

750 done += len(chunk) 

751 typer.echo(f" {done:>6,} / {len(ids):,}") 

752 

753 with session_scope() as session: 

754 left, left_bytes = session.execute( 

755 text("select count(*), coalesce(sum(length(body)), 0) from raw.raw_payload") 

756 ).one() 

757 typer.secho( 

758 f"{done:,} payloads pruned, {left:,} left ({_human(int(left_bytes))} logical)", 

759 fg="green", 

760 bold=True, 

761 )

764@db_app.command("reset-raw") 

765def db_reset_raw( 

766 yes: bool = typer.Option(False, "--yes", help="Required. There is no undo."), 

767) -> None: 

768 """Empty the raw payload store outright, when the database is too full to prune. 

769 

770 This exists because of a genuine deadlock. The routine sweep deletes the 

771 surplus in batches, and a delete has to *write*: dead tuples are recorded 

772 rather than merely dropped, and evidence_unit.raw_payload_id is ON DELETE 

773 SET NULL, so every payload removed also rewrites the evidence rows pointing 

774 at it. On a database that is already at its ceiling there is nowhere to put 

775 any of that, and the sweep fails with the same DiskFull it was written to 

776 cure. Fifty rows at a time did not help, because the problem is not the 

777 size of the batch, it is that no write of any size can be recorded. 

778 

779 TRUNCATE is the way out, because it allocates nothing: it drops the whole 

780 relation and creates an empty one. The foreign key is what normally forbids 

781 it, and TRUNCATE ... CASCADE would take evidence_unit with it, which is the 

782 product rather than the receipt. So the constraint comes off first, the 

783 table is emptied, the now-dangling pointers are nulled while there is room 

784 to do it, and the constraint goes back on. 

785 

786 What is lost is every stored HTTP response, including the recent ones the 

787 retention window would have kept. That is affordable only because nothing 

788 reads them: parse() works on the payload in memory during the same run, and 

789 no replay path exists yet. Every evidence unit, verdict and citation is 

790 untouched. 

791 

792 Not wired into the nightly, and it should not be. This is a recovery for a 

793 database that has already stopped, and the ordinary sweep plus compression 

794 is what stops it happening again. 

795 """ 

796 if not yes: 

797 typer.secho("refusing without --yes. This empties raw.raw_payload completely.", fg="red") 

798 raise typer.Exit(1) 

799 

800 engine = get_engine() 

801 with session_scope() as session: 

802 before, before_bytes = session.execute( 

803 text("select count(*), coalesce(sum(length(body)), 0) from raw.raw_payload") 

804 ).one() 

805 pointing = ( 

806 session.scalar( 

807 text("select count(*) from core.evidence_unit where raw_payload_id is not null") 

808 ) 

809 or 0 

810 ) 

811 # By name from the catalogue rather than a guess at the convention, so 

812 # this keeps working if the constraint is ever renamed. 

813 constraints = [ 

814 row[0] 

815 for row in session.execute( 

816 text( 

817 """ 

818 select con.conname 

819 from pg_constraint con 

820 join pg_class child on child.oid = con.conrelid 

821 join pg_namespace cn on cn.oid = child.relnamespace 

822 join pg_class parent on parent.oid = con.confrelid 

823 join pg_namespace pn on pn.oid = parent.relnamespace 

824 where con.contype = 'f' 

825 and pn.nspname = 'raw' and parent.relname = 'raw_payload' 

826 and cn.nspname = 'core' and child.relname = 'evidence_unit' 

827 """ 

828 ) 

829 ).all() 

830 ] 

831 

832 typer.echo(f" payloads held {before:>8,} ({_human(int(before_bytes))} logical)") 

833 typer.echo(f" evidence pointing {pointing:>8,}") 

834 typer.echo(f" constraints to lift {', '.join(constraints) or 'none'}") 

835 

836 # Autocommit throughout: TRUNCATE inside a transaction holds the old 

837 # relation until commit, which defeats the entire point of using it on a 

838 # database that has no room to hold anything. 

839 with engine.connect().execution_options(isolation_level="AUTOCOMMIT") as conn: 

840 for name in constraints: 

841 conn.execute(text(f"alter table core.evidence_unit drop constraint {name}")) 

842 typer.echo(" constraint lifted") 

843 

844 conn.execute(text("truncate table raw.raw_payload")) 

845 typer.secho(" raw.raw_payload emptied", fg="green") 

846 

847 # Only now is there room to write anything, which is why this is here 

848 # and not before the truncate. 

849 conn.execute( 

850 text( 

851 "update core.evidence_unit set raw_payload_id = null where raw_payload_id is not null" 

852 ) 

853 ) 

854 typer.echo(" dangling pointers cleared") 

855 

856 for name in constraints: 

857 conn.execute( 

858 text( 

859 f"alter table core.evidence_unit add constraint {name} " 

860 "foreign key (raw_payload_id) references raw.raw_payload(id) on delete set null" 

861 ) 

862 ) 

863 typer.echo(" constraint restored") 

864 

865 conn.execute(text("vacuum (analyze) core.evidence_unit")) 

866 

867 typer.secho( 

868 f"{before:,} payloads dropped. Evidence, verdicts and citations untouched.", 

869 fg="green", 

870 bold=True, 

871 )

1055@pipeline_app.command("nightly") 

1056def pipeline_nightly( 

1057 sources: str = typer.Option(DEFAULT_SOURCES, "--sources", help="Comma-separated source keys."), 

1058 limit_variants: int | None = typer.Option(None, "--limit", help="Only the first N variants."), 

1059) -> None: 

1060 """Everything, in order. This is what the scheduled workflow calls. 

1061 

1062 A source that fails is reported and skipped. The stages after it still 

1063 run, because the product is meant to stay complete with three of eight 

1064 sources alive. 

1065 """ 

1066 logging.basicConfig(level=logging.WARNING, format="%(levelname)s %(name)s: %(message)s") 

1067 overall = time.monotonic() 

1068 

1069 with session_scope() as session: 

1070 seed_all(session) 

1071 seed_catalogue(session) 

1072 

1073 requested = [s.strip() for s in sources.split(",") if s.strip()] 

1074 failed: list[str] = [] 

1075 for key in requested: 

1076 try: 

1077 connector = registry.get(key) 

1078 except KeyError as exc: 

1079 typer.secho(f" {key}: {exc}", fg="red") 

1080 failed.append(key) 

1081 continue 

1082 with session_scope() as session: 

1083 result = run_connector(session, connector, limit_variants=limit_variants) 

1084 colour = "green" if result.ok else "yellow" 

1085 typer.secho( 

1086 f" ingest {key:16} {result.status.value:14} " 

1087 f"+{result.units_inserted:,} units, {result.units_skipped:,} skipped", 

1088 fg=colour, 

1089 ) 

1090 if not result.ok: 

1091 failed.append(key) 

1092 

1093 for name, fn in ( 

1094 ("resolve", resolve_listings), 

1095 ("extract", extract_opinions), 

1096 ("score", score_credibility), 

1097 ): 

1098 start = time.monotonic() 

1099 with session_scope() as session: 

1100 stats = fn(session) 

1101 _report(name, stats, time.monotonic() - start) 

1102 

1103 start = time.monotonic() 

1104 with session_scope() as session: 

1105 stats = fuse_all(session, variant_limit=limit_variants) 

1106 _report("fuse", stats, time.monotonic() - start) 

1107 

1108 typer.secho(f"\nnightly finished in {time.monotonic() - overall:.1f}s", bold=True) 

1109 if failed: 

1110 typer.secho(f"sources that did not succeed: {', '.join(failed)}", fg="yellow") 

1111 

1112 # Losing some sources is the resilience contract working as designed. 

1113 # Losing all of them is a broken run and must not exit zero: every stage 

1114 # downstream will happily "succeed" over an empty database and suppress 

1115 # every verdict, which looks like a healthy run from the outside. 

1116 if requested and len(failed) == len(requested): 

1117 typer.secho( 

1118 f"every requested source failed ({', '.join(requested)}). "
