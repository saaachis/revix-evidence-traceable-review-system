1"""A deterministic development fixture. 

2 

3This is NOT a real source and it is labelled as such everywhere it appears. 

4It exists for two reasons the project genuinely needs: 

5 

6 1. Tests need input that does not change between runs. Asserting on scraped 

7 text would make the suite fail whenever a website edits a page. 

8 2. Every connector needs a documented fallback, per proposal section 22, so 

9 that a blocked source degrades coverage rather than stopping work. 

10 

11The text is generated from templates rather than copied from any site, so 

12nothing here is anyone else's writing. The covariate structure is deliberate: 

13automatic owners are less happy about the gearbox, smaller-city owners are 

14less happy about service, and early build years are less happy about 

15reliability. That gives the divergence analysis something real to find, and it 

16mirrors the patterns the literature says exist. 

17"""

127class FixtureConnector: 

128 """Generates deterministic synthetic evidence for one variant. 

129 

130 Parameterised by source kind so that several can be registered. That is 

131 not padding: a single source makes the source-weighted strategy 

132 meaningless, because there is nothing to weight differently, and it can 

133 never clear the three-source evidence floor. Three fixtures standing in 

134 for owner reviews, a forum and an expert publication exercise both. 

135 """ 

136 

137 base_url: str | None = None 

138 robots_policy: str | None = "Not applicable. Generated locally, nothing is fetched." 

139 rate_limit_rpm: int = 6000 

140 

141 def __init__( 

142 self, 

143 *, 

144 source_key: str = SOURCE_KEY, 

145 display_name: str | None = None, 

146 kind: SourceKind = SourceKind.DATASET, 

147 source_prior: float = 0.5, 

148 per_variant: int = 60, 

149 verified_rate: float = 0.62, 

150 detail_bias: float = 0.0, 

151 ) -> None: 

152 self.source_key = source_key 

153 self.display_name = display_name or f"Development fixture: {source_key} (synthetic)" 

154 self.kind = kind 

155 self.default_source_prior = source_prior 

156 self.per_variant = per_variant 

157 self.verified_rate = verified_rate 

158 # Shifts how positive this source runs. Expert publications review 

159 # pre-production cars for a weekend and come out kinder than owners, 

160 # which is the media-versus-owner gap the verdict page reports. 

161 self.detail_bias = detail_bias 

162 

163 def discover(self, seed: CatalogSeed) -> Iterable[ExternalRef]: 

164 yield ExternalRef( 

165 external_id=f"{self.source_key}:{seed.variant_id}", 

166 url=f"{self.source_key}://{seed.variant_id}", 

167 seed=seed, 

168 ) 

169 

170 def fetch(self, ref: ExternalRef) -> RawPayload: 

171 # Nothing is fetched. The payload records what was asked for, so the 

172 # raw store still has a replayable record of every generated batch. 

173 body = f"{self.source_key} request for {ref.external_id}".encode() 

174 return RawPayload(ref=ref, body=body, fetched_at=datetime.now(UTC), http_status=200) 

175 

176 def parse(self, raw: RawPayload) -> list[EvidenceUnitDraft]: 

177 seed = raw.ref.seed 

178 if seed is None: # pragma: no cover - discover always attaches one 

179 return [] 

180 return list(self._generate(seed)) 

181 

182 def _generate(self, seed: CatalogSeed) -> Iterable[EvidenceUnitDraft]: 

183 # Seeded by variant id, so the same variant always produces the same 

184 # corpus and tests can assert on exact numbers. 

185 rng = random.Random(f"revix:{self.source_key}:{seed.variant_id}") 

186 name = seed.variant_name.casefold() 

187 is_automatic = any(t in name for t in (" at", "dct", "cvt", "amt", "ivt", "dsg", "dca")) 

188 now = datetime.now(UTC) 

189 

190 for i in range(self.per_variant): 

191 months = rng.choice([1, 2, 3, 6, 9, 12, 18, 24, 30, 36, 44, 52]) 

192 km = int(months * rng.uniform(600, 1800)) 

193 small_city = rng.random() < 0.4 

194 city = rng.choice(CITIES_SMALL if small_city else CITIES_METRO) 

195 early_year = rng.random() < 0.35 

196 verified = rng.random() < self.verified_rate 

197 

198 bias = self.detail_bias 

199 if is_automatic: 

200 bias += COVARIATE_EFFECTS["automatic"]["engine_gearbox"] * 0.2 

201 aspects = rng.sample(sorted(POSITIVE), k=rng.randint(2, 4)) 

202 

203 sentences: list[str] = [] 

204 polarities: list[float] = [] 

205 for aspect in aspects: 

206 base = rng.gauss(0.45, 0.45) 

207 if aspect == "engine_gearbox" and is_automatic: 

208 base += COVARIATE_EFFECTS["automatic"]["engine_gearbox"] 

209 if aspect == "service_aftersales" and small_city: 

210 base += COVARIATE_EFFECTS["small_city"]["service_aftersales"] 

211 if aspect == "long_term_reliability" and early_year: 

212 base += COVARIATE_EFFECTS["early_year"]["long_term_reliability"] 

213 base = max(-1.0, min(1.0, base + bias)) 

214 pool = POSITIVE[aspect] if base >= 0 else NEGATIVE[aspect] 

215 sentences.append(rng.choice(pool)) 

216 polarities.append(base) 

217 

218 mean_polarity = sum(polarities) / len(polarities) 

219 opening = f"Owned for {months} months, {km:,} km, driving mostly in {city}." 

220 text = " ".join([opening, *sentences]) 

221 

222 yield EvidenceUnitDraft( 

223 external_id=f"{self.source_key}:{seed.variant_id}:{i:04d}", 

224 text=text, 

225 modality=Modality.TEXT, 

226 url=f"{self.source_key}://{seed.variant_id}/{i:04d}", 

227 author_ref=f"fx-{rng.getrandbits(48):012x}", 

228 lang="en", 

229 published_at=now - timedelta(days=rng.randint(10, 1200)), 

230 rating_raw=round(min(5.0, max(1.0, 3.0 + mean_polarity * 2)), 1), 

231 rating_scale_max=5.0, 

232 is_verified_owner=verified, 

233 helpful_votes=rng.randint(0, 40), 

234 total_votes=rng.randint(40, 80), 

235 ownership_duration_months=months, 

236 km_driven=km, 

237 listing_title=f"{seed.manufacturer} {seed.model} {seed.variant_name}", 

238 variant_hint=seed.variant_name, 

239 model_hint=seed.model, 

240 )
